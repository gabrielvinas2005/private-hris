INSERT INTO leave_credits (
[employee_id]
,[leave_type_id]
,[credits]
,[created_at]
,[updated_at]
)
SELECT 
b.id as [employee_id]
,a.leave_type as [leave_type_id]
,a.credits as [credits]
,GETDATE() as [created_at]
,null as [updated_at]
FROM mig_leave_credits as a
INNER JOIN employees as b ON a.employee_no = b.employee_no
